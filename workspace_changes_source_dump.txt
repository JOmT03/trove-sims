<?php

namespace App\Http\Controllers;

use App\Models\BranchOrder;
use App\Models\BranchOrderItem;
use App\Models\Product;
use App\Models\Site;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class BranchOrderController extends Controller
{
    public function index()
    {
        $open = BranchOrder::decorate(
            BranchOrder::with(['items.product', 'destinationSite'])
                ->whereIn('status', ['requested', 'produced', 'partial'])
                ->orderBy('needed_by')->get()
        );

        $recent = BranchOrder::decorate(
            BranchOrder::with(['items.product', 'destinationSite'])
                ->whereIn('status', ['complete', 'cancelled'])
                ->orderBy('updated_at', 'desc')->limit(15)->get()
        );

        $products = Product::where('status', 'active')->orderBy('product_name')->get();
        $sites = Site::orderBy('site_name')->get();

        return redirect()->route('branch-transfers.index');
    }

    public function store(Request $request)
    {
        $v = $request->validate([
            'destination_site_id' => 'nullable|exists:sites,id',
            'needed_by'           => 'required|date',
            'notes'               => 'nullable|string|max:500',
            'items'               => 'required|array|min:1',
            'items.*.product_id'  => 'required|exists:products,id',
            'items.*.qty'         => 'required|integer|min:1',
        ]);

        DB::transaction(function () use ($v) {
            $order = BranchOrder::create([
                'destination_site_id' => $v['destination_site_id'] ?? null,
                'needed_by'           => $v['needed_by'],
                'notes'               => $v['notes'] ?? null,
                'status'              => 'requested',
                'created_by'          => auth()->id(),
            ]);
            foreach ($v['items'] as $it) {
                $order->items()->create(['product_id' => $it['product_id'], 'qty' => $it['qty']]);
            }
            $this->audit('branch_order.requested', $order, ['needed_by' => $v['needed_by'], 'items' => count($v['items'])]);
        });

        return redirect()->route('branch-transfers.index')->with('success', 'Order recorded.');
    }

    public function produce(BranchOrder $order)
    {
        if ($order->status === 'requested') {
            $order->update(['status' => 'produced']);
            $this->audit('branch_order.produced', $order, []);
        }
        return back()->with('success', 'Order marked as produced.');
    }

    public function cancel(BranchOrder $order)
    {
        if (in_array($order->status, ['requested', 'produced'], true)) {
            $order->update(['status' => 'cancelled']);
            $this->audit('branch_order.cancelled', $order, []);
            return back()->with('success', 'Order cancelled.');
        }
        return back()->with('error', 'Only orders with nothing sent yet can be cancelled. Use Close short instead.');
    }

    /** Stop waiting for the remaining balance. The balance is not sold and earns no revenue. */
    public function closeShort(Request $request, BranchOrder $order)
    {
        $v = $request->validate(['reason' => 'required|string|min:3|max:500']);

        if (!in_array($order->status, ['requested', 'produced', 'partial'], true)) {
            return back()->with('error', 'This order is already closed.');
        }

        $fresh = BranchOrder::with('items')->find($order->id);
        BranchOrder::decorate(collect([$fresh]));
        $short = $fresh->balance_qty;

        DB::transaction(function () use ($fresh, $v) {
            foreach ($fresh->items as $it) {
                if ($it->balance > 0) {
                    BranchOrderItem::whereKey($it->id)->update(['qty_closed' => (int) $it->qty_closed + $it->balance]);
                }
            }
            BranchOrder::whereKey($fresh->id)->update(['status' => 'complete', 'short_reason' => $v['reason']]);
        });

        $this->audit('branch_order.closed_short', $order, ['short_qty' => $short, 'reason' => $v['reason']]);

        return back()->with('success', "Order closed. {$short} pcs not delivered.");
    }

    private function audit(string $action, $model, array $data): void
    {
        try {
            if (class_exists(\App\Models\FinancialAuditLog::class)) {
                \App\Models\FinancialAuditLog::record($action, $model, $data);
            }
        } catch (\Throwable $e) {
            \Log::warning('Audit log failed: ' . $e->getMessage());
        }
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Batch;
use App\Models\BatchItem;
use App\Models\BranchOrder;
use App\Models\Product;
use App\Models\Site;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class BranchTransferController extends Controller
{
    public function index()
    {
        $batches = Batch::with(['items', 'sourceSite', 'destinationSite'])
            ->orderBy('batch_date', 'desc')
            ->get();

        $orders = BranchOrder::decorate(
            BranchOrder::with(['items.product', 'destinationSite'])
                ->orderBy('needed_by', 'desc')->limit(20)->get()
        );

        return $this->board($batches);
    }

    private function board($batches)
    {
        $pending = BranchOrder::with(['items.product', 'destinationSite'])
            ->whereIn('status', ['requested', 'produced', 'partial'])
            ->orderBy('needed_by')->get();
        $done = BranchOrder::with(['items.product', 'destinationSite'])
            ->whereIn('status', ['complete', 'cancelled'])
            ->orderBy('needed_by', 'desc')->limit(10)->get();
        $orders = BranchOrder::decorate($pending->concat($done));

        // Note on the latest live batch of each order that is still short (or was closed short)
        $linked = $batches->filter(fn ($b) => $b->branch_order_id && empty($b->voided_at))
            ->sortByDesc(fn ($b) => $b->batch_date->format('Y-m-d') . str_pad((string) $b->id, 10, '0', STR_PAD_LEFT));
        $byOrder = BranchOrder::decorate(
            BranchOrder::with('items')->whereIn('id', $linked->pluck('branch_order_id')->unique()->all())->get()
        )->keyBy('id');

        $notes = [];
        $seen = [];
        foreach ($linked as $b) {
            $oid = $b->branch_order_id;
            if (isset($seen[$oid])) {
                continue;
            }
            $seen[$oid] = true;
            $o = $byOrder[$oid] ?? null;
            if (!$o) {
                continue;
            }
            if ($o->balance_qty > 0) {
                $notes[$b->id] = "Short {$o->balance_qty} (order #{$oid}): to be delivered in the next batch";
            } elseif ($o->short_reason) {
                $notes[$b->id] = 'Closed short ' . $o->items->sum('qty_closed') . " (order #{$oid}): " . $o->short_reason;
            }
        }

        $products = Product::where('status', 'active')->orderBy('product_name')->get();
        $sites = Site::orderBy('site_name')->get();

        return view('branch-transfers.index', compact('batches', 'orders', 'notes', 'products', 'sites'));
    }

    public function create(Request $request)
    {
        $sites = Site::orderBy('site_name')->get();
        $products = Product::where('status', 'active')->orderBy('product_name')->get();

        $orders = BranchOrder::decorate(
            BranchOrder::with('items.product')
                ->whereIn('status', ['requested', 'produced', 'partial'])
                ->orderBy('needed_by')->get()
        )->filter(fn ($o) => $o->balance_qty > 0)->values();

        $selected = $request->query('order');

        return view('branch-transfers.create', compact('sites', 'products', 'orders', 'selected'));
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'source_site_id'      => 'nullable|exists:sites,id',
            'destination_site_id' => 'nullable|exists:sites,id',
            'branch_order_id'     => 'nullable|exists:branch_orders,id',
            'batch_date'          => 'required|date',
            'items'               => 'required|array|min:1',
            'items.*.product_id'  => 'required|exists:products,id',
            'items.*.qty_sent'    => 'required|integer|min:1',
            'items.*.unit_price'  => 'required|numeric|min:0',
        ]);

        try {
            DB::beginTransaction();

            foreach ($validated['items'] as $item) {
                $product = Product::findOrFail($item['product_id']);
                if ($item['qty_sent'] > $product->stock_quantity) {
                    throw new \Exception(
                        "Not enough {$product->product_name}: only {$product->stock_quantity} on hand, but you tried to send {$item['qty_sent']}."
                    );
                }
            }

            $batch = Batch::create([
                'source_site_id'      => $validated['source_site_id'] ?? null,
                'destination_site_id' => $validated['destination_site_id'] ?? null,
                'batch_date'          => $validated['batch_date'],
                'status'              => 'sent',
                'created_by'          => auth()->id(),
            ]);

            foreach ($validated['items'] as $item) {
                BatchItem::create([
                    'batch_id'     => $batch->id,
                    'product_id'   => $item['product_id'],
                    'qty_sent'     => $item['qty_sent'],
                    'qty_returned' => 0,
                    'unit_price'   => $item['unit_price'],
                ]);

                Product::where('id', $item['product_id'])->decrement('stock_quantity', $item['qty_sent']);
            }

            if (!empty($validated['branch_order_id'])) {
                DB::table('batches')->where('id', $batch->id)->update(['branch_order_id' => $validated['branch_order_id']]);
                $order = BranchOrder::find($validated['branch_order_id']);
                if ($order) {
                    $order->refreshStatus();
                }
            }

            $this->audit('branch_transfer.sent', $batch, [
                'batch_date' => $validated['batch_date'],
                'items'      => count($validated['items']),
                'order_id'   => $validated['branch_order_id'] ?? null,
            ]);

            DB::commit();

            return redirect()->route('branch-transfers.show', $batch)
                ->with('success', 'Batch sent. This is a final sale and was counted as revenue.');
        } catch (\Exception $e) {
            DB::rollBack();
            \Log::error('Batch send failed: ' . $e->getMessage());
            return back()->withInput()->with('error', $e->getMessage());
        }
    }

    public function show(Batch $batch)
    {
        $batch->load(['items.product', 'sourceSite', 'destinationSite', 'creator']);
        $voidInfo = DB::table('batches')->where('id', $batch->id)->first(['voided_at', 'void_reason', 'branch_order_id']);
        return view('branch-transfers.show', compact('batch', 'voidInfo'));
    }

    /** Undo a mistaken send: restore stock, exclude from revenue. Reason required. */
    public function void(Request $request, Batch $batch)
    {
        $request->validate(['reason' => 'required|string|min:3|max:500']);

        $row = DB::table('batches')->where('id', $batch->id)->first();
        if ($row->voided_at) {
            return back()->with('error', 'This batch is already voided.');
        }
        if (\App\Models\PeriodClosure::isClosed($batch->batch_date)) {
            return back()->with('error', 'This batch belongs to a closed month and cannot be voided.');
        }

        try {
            DB::beginTransaction();

            foreach ($batch->items as $item) {
                if ($item->product_id) {
                    Product::where('id', $item->product_id)->increment('stock_quantity', $item->qty_sent);
                }
            }

            DB::table('batches')->where('id', $batch->id)->update([
                'voided_at'   => now(),
                'void_reason' => $request->reason,
                'voided_by'   => auth()->id(),
            ]);

            if ($row->branch_order_id) {
                $order = BranchOrder::find($row->branch_order_id);
                if ($order) {
                    $order->refreshStatus();
                }
            }

            $this->audit('branch_transfer.voided', $batch, ['reason' => $request->reason]);

            DB::commit();

            return redirect()->route('branch-transfers.show', $batch)
                ->with('success', 'Batch voided. Stock restored and revenue removed.');
        } catch (\Exception $e) {
            DB::rollBack();
            \Log::error('Batch void failed: ' . $e->getMessage());
            return back()->with('error', 'Failed to void: ' . $e->getMessage());
        }
    }

    private function audit(string $action, $model, array $data): void
    {
        try {
            if (class_exists(\App\Models\FinancialAuditLog::class)) {
                \App\Models\FinancialAuditLog::record($action, $model, $data);
            }
        } catch (\Throwable $e) {
            \Log::warning('Audit log failed: ' . $e->getMessage());
        }
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Product;
use App\Models\Sale;
use App\Models\SalePayment;
use App\Models\DayClosure;
use App\Models\Site;
use App\Models\FinancialAuditLog;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

/**
 * POS â€” counter sale of FINISHED products (walk-in).
 * Deducts finished stock (Product.stock_quantity) only. Raw materials are NOT
 * touched here: they were already consumed when the product was produced.
 * A sale is a ledger entry â€” it is voided (with a reason), never deleted.
 */
class PosController extends Controller
{
    public function index()
    {
        $products = Product::where('status', 'active')
            ->orderBy('product_name')
            ->get(['id', 'product_name', 'category', 'price', 'stock_quantity', 'image_path']);

        $recent = Sale::with('cashier')
            ->orderByDesc('id')
            ->limit(8)
            ->get();

        return view('pos.index', compact('products', 'recent'));
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'items'                => 'required|array|min:1',
            'items.*.product_id'   => 'required|exists:products,id',
            'items.*.quantity'     => 'required|integer|min:1',
            'payments'             => 'required|array|min:1',
            'payments.*.method'    => 'required|in:' . implode(',', SalePayment::METHODS),
            'payments.*.amount'    => 'required|numeric|min:0.01',
            'payments.*.reference' => 'nullable|string|max:100',
        ]);

        // Digital payments must carry a reference number.
        foreach ($data['payments'] as $p) {
            if ($p['method'] !== 'cash' && blank($p['reference'] ?? null)) {
                return back()->withInput()
                    ->with('error', 'A reference number is required for ' . strtoupper($p['method']) . ' payments.');
            }
        }

        try {
            $sale = DB::transaction(function () use ($data) {
                // Lock the products we are about to sell so two cashiers cannot
                // oversell the same stock at the same time.
                $ids = collect($data['items'])->pluck('product_id')->all();
                $products = Product::whereIn('id', $ids)->lockForUpdate()->get()->keyBy('id');

                $total = 0;
                $lines = [];
                foreach ($data['items'] as $row) {
                    $product = $products[$row['product_id']];
                    $qty = (int) $row['quantity'];
                    if ((float) $product->stock_quantity < $qty) {
                        throw new \RuntimeException(
                            "Not enough stock for {$product->product_name} (have {$product->stock_quantity}, need {$qty})."
                        );
                    }
                    $lineTotal = round((float) $product->price * $qty, 2);
                    $total += $lineTotal;
                    $lines[] = [$product, $qty, $lineTotal];
                }

                $paid = round(collect($data['payments'])->sum(fn ($p) => (float) $p['amount']), 2);
                if ($paid + 0.009 < $total) {
                    throw new \RuntimeException(
                        'Payments (' . number_format($paid, 2) . ') are less than the total (' . number_format($total, 2) . ').'
                    );
                }

                $siteId = auth()->user()->site_id ?? Site::query()->value('id');

                $sale = Sale::create([
                    'sale_number'  => 'S-' . now()->format('ymd') . '-' . strtoupper(substr(uniqid(), -5)),
                    'site_id'      => $siteId,
                    'cashier_id'   => auth()->id(),
                    'total'        => $total,
                    'status'       => 'completed',
                    'completed_at' => now(),
                ]);

                foreach ($lines as [$product, $qty, $lineTotal]) {
                    $sale->items()->create([
                        'product_id'   => $product->id,
                        'product_name' => $product->product_name,
                        'quantity'     => $qty,
                        'unit_price'   => $product->price,
                        'line_total'   => $lineTotal,
                    ]);
                    $product->decrement('stock_quantity', $qty); // finished stock only
                }

                foreach ($data['payments'] as $p) {
                    if ((float) $p['amount'] <= 0) {
                        continue;
                    }
                    $sale->payments()->create([
                        'method'    => $p['method'],
                        'amount'    => round((float) $p['amount'], 2),
                        'reference' => $p['method'] === 'cash' ? null : ($p['reference'] ?? null),
                    ]);
                }

                FinancialAuditLog::record('sale.recorded', $sale, [
                    'total'    => (string) $sale->total,
                    'items'    => count($lines),
                    'payments' => collect($data['payments'])
                        ->map(fn ($p) => $p['method'] . ':' . $p['amount'])->implode(', '),
                ]);

                return $sale;
            });
        } catch (\Throwable $e) {
            return back()->withInput()->with('error', $e->getMessage());
        }

        return redirect()->route('pos.index')
            ->with('success', 'Sale ' . $sale->sale_number . ' recorded. Total: ' . number_format($sale->total, 2) . '.');
    }

    public function void(Request $request, Sale $sale)
    {
        $data = $request->validate(['reason' => 'required|string|max:255']);

        if ($sale->isVoided()) {
            return back()->with('error', 'That sale is already voided.');
        }
        if (DayClosure::isClosed($sale->completed_at)) {
            return back()->with('error', 'That sale falls on a day that is already closed. It can no longer be voided.');
        }

        DB::transaction(function () use ($sale, $data) {
            foreach ($sale->items as $item) {
                if ($item->product_id) {
                    $product = Product::where('id', $item->product_id)->lockForUpdate()->first();
                    $product?->increment('stock_quantity', $item->quantity); // put the stock back
                }
            }

            $sale->update([
                'status'      => 'voided',
                'voided_at'   => now(),
                'voided_by'   => auth()->id(),
                'void_reason' => $data['reason'],
            ]);

            FinancialAuditLog::record('sale.voided', $sale, [
                'reason' => $data['reason'],
                'total'  => (string) $sale->total,
            ]);
        });

        return back()->with('success', 'Sale ' . $sale->sale_number . ' voided and stock restored.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Product;
use Illuminate\Http\Request;

/**
 * Production Capacity. Reads each product's recipe (product_raw_materials:
 * quantity_needed per unit) against current raw-material stock, and shows how
 * many units of each product can still be produced right now, plus the
 * ingredient that runs out first (the limiting ingredient). This lets the user
 * see producible output straight from inventory, instead of recording a batch
 * first and only then learning what stock allows.
 */
class ProductionCapacityController extends Controller
{
    public function index(Request $request)
    {
        $search = trim((string) $request->get('search', ''));

        $products = Product::with('materials')
            ->when($search !== '', fn ($q) => $q->where('product_name', 'like', '%' . $search . '%'))
            ->orderBy('product_name')
            ->get();

        $rows = [];
        foreach ($products as $p) {
            if (strtolower((string) $p->status) === 'archived') {
                continue;
            }

            $materials = $p->materials;
            if ($materials->isEmpty()) {
                $rows[] = ['product' => $p, 'producible' => null, 'limiting' => null, 'recipe' => []];
                continue;
            }

            $minUnits = null;
            $limiting = null;
            $recipe   = [];

            foreach ($materials as $m) {
                $needed    = (float) $m->pivot->quantity_needed;
                $available = (float) $m->quantity_on_hand - (float) $m->quantity_damaged;
                if ($available < 0) {
                    $available = 0.0;
                }
                $canMake = $needed > 0 ? (int) floor($available / $needed) : null;

                $recipe[] = [
                    'name'      => $m->item_name,
                    'unit'      => $m->unit,
                    'needed'    => $needed,
                    'available' => $available,
                    'canMake'   => $canMake,
                ];

                if ($canMake !== null && ($minUnits === null || $canMake < $minUnits)) {
                    $minUnits = $canMake;
                    $limiting = $m->item_name;
                }
            }

            $rows[] = [
                'product'    => $p,
                'producible' => $minUnits,
                'limiting'   => $limiting,
                'recipe'     => $recipe,
            ];
        }

        return view('inventory.capacity', compact('rows', 'search'));
    }
}

===== app/Http/Controllers/ProductionController.php =====
<?php

namespace App\Http\Controllers;

use App\Models\Inventory;
use App\Models\InventoryLog;
use App\Models\Product;
use App\Models\ProductionAudit;
use App\Models\RecipeVersion;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Gate;

class ProductionController extends Controller
{
    /**
     * Yield preview: for every product with an active recipe, how many units
     * the CURRENT raw-material stock can produce (and which ingredient limits it).
     */
    private function yieldRows(): array
    {
        $versions = RecipeVersion::where('is_active', true)
            ->where('is_archived', false)
            ->get()
            ->keyBy('product_id');

        if ($versions->isEmpty()) {
            return [];
        }

        $products = Product::whereIn('id', $versions->keys()->all())->orderBy('product_name')->get();

        $invIds = [];
        foreach ($versions as $v) {
            foreach (($v->items ?? []) as $it) {
                if (!empty($it['inventory_id'])) {
                    $invIds[] = (int) $it['inventory_id'];
                }
            }
        }
        $inv = Inventory::whereIn('id', array_unique($invIds))->get()->keyBy('id');

        $rows = [];
        foreach ($products as $p) {
            $v = $versions[$p->id];
            $lines = [];
            $max = null;
            $limit = null;

            foreach (($v->items ?? []) as $it) {
                $i   = $inv[(int) ($it['inventory_id'] ?? 0)] ?? null;
                $per = (float) ($it['quantity_used'] ?? 0);
                if (!$i || $per <= 0) {
                    continue;
                }
                $on  = (float) $i->quantity_on_hand;
                $can = max(0, (int) floor(round($on / $per, 6)));

                $lines[] = [
                    'name' => $i->item_name,
                    'unit' => $i->unit,
                    'on'   => $on,
                    'per'  => $per,
                    'can'  => $can,
                ];

                if ($max === null || $can < $max) {
                    $max   = $can;
                    $limit = $i->item_name;
                }
            }

            if (empty($lines)) {
                continue;
            }

            $rows[$p->id] = [
                'id'     => $p->id,
                'name'   => $p->product_name,
                'recipe' => $v->name,
                'stock'  => (int) $p->stock_quantity,
                'max'    => (int) $max,
                'limit'  => $limit,
                'items'  => $lines,
            ];
        }

        return $rows;
    }

    /** Inventory > Production tab (yield preview + record form + audit log). */
    public function create(Request $request)
    {
        $yield   = $this->yieldRows();
        $isAdmin = Gate::allows('admin');

        $audits = null;
        $stats  = ['runs' => 0, 'flagged' => 0, 'missing' => 0];

        if ($isAdmin) {
            $since = now()->subDays(30);

            $audits = ProductionAudit::with('user')
                ->when($request->get('flag') === 'variance', fn ($q) => $q->where('variance_qty', '!=', 0))
                ->orderByDesc('id')
                ->simplePaginate(15)
                ->withQueryString();

            $stats = [
                'runs'    => ProductionAudit::where('created_at', '>=', $since)->count(),
                'flagged' => ProductionAudit::where('created_at', '>=', $since)->where('variance_qty', '!=', 0)->count(),
                'missing' => (int) ProductionAudit::where('created_at', '>=', $since)->where('variance_qty', '>', 0)->sum('variance_qty'),
            ];
        }

        $flag = $request->get('flag') === 'variance' ? 'variance' : 'all';

        return view('inventory.production', compact('yield', 'isAdmin', 'audits', 'stats', 'flag'));
    }

    /** Record a production run. Expected yield is computed by the server (recipe + current stock). */
    public function store(Request $request)
    {
        abort_unless(Gate::allows('admin'), 403);

        $data = $request->validate([
            'product_id' => 'required|exists:products,id',
            'actual_qty' => 'required|integer|min:0',
            'notes'      => 'nullable|string|max:500',
        ]);

        $actual = (int) $data['actual_qty'];
        $notes  = trim((string) ($data['notes'] ?? ''));

        try {
            $audit = DB::transaction(function () use ($data, $actual, $notes) {
                $product = Product::lockForUpdate()->findOrFail($data['product_id']);

                $version = RecipeVersion::where('product_id', $product->id)
                    ->where('is_active', true)
                    ->where('is_archived', false)
                    ->first();

                if (!$version || empty($version->items)) {
                    throw new \RuntimeException('This product has no active recipe.');
                }

                // Server computes the expected yield from CURRENT stock + recipe
                $lines    = [];
                $expected = null;
                foreach ($version->items as $it) {
                    $inv = Inventory::lockForUpdate()->find((int) ($it['inventory_id'] ?? 0));
                    $per = (float) ($it['quantity_used'] ?? 0);
                    if (!$inv || $per <= 0) {
                        continue;
                    }
                    $can      = max(0, (int) floor(round(((float) $inv->quantity_on_hand) / $per, 6)));
                    $expected = $expected === null ? $can : min($expected, $can);
                    $lines[]  = [$inv, $per];
                }

                if (empty($lines) || $expected < 1) {
                    throw new \RuntimeException('Current stock cannot produce even 1 unit of this product.');
                }
                if ($actual !== $expected && $notes === '') {
                    throw new \RuntimeException("Inventory says {$expected} should be produced, but you entered {$actual}. Please write the reason for the difference.");
                }

                // Materials issued for the full expected batch
                $snapshot = [];
                foreach ($lines as [$inv, $per]) {
                    $need   = round($per * $expected, 2);
                    $before = (float) $inv->quantity_on_hand;
                    $after  = round($before - $need, 2);

                    $inv->update(['quantity_on_hand' => $after]);

                    InventoryLog::create([
                        'inventory_id' => $inv->id,
                        'type'         => 'used',
                        'quantity'     => $need,
                        'reference'    => 'PRODUCTION-AUDIT',
                        'notes'        => "Issued for {$expected} x {$product->product_name} (actual produced: {$actual})",
                        'user_id'      => auth()->id(),
                    ]);

                    $snapshot[] = [
                        'inventory_id' => $inv->id,
                        'name'         => $inv->item_name,
                        'unit'         => $inv->unit,
                        'before'       => $before,
                        'used'         => $need,
                        'after'        => $after,
                    ];
                }

                $product->update(['stock_quantity' => (int) $product->stock_quantity + $actual]);

                $audit = ProductionAudit::create([
                    'product_id'        => $product->id,
                    'product_name'      => $product->product_name,
                    'recipe_version_id' => $version->id,
                    'recipe_name'       => $version->name,
                    'expected_qty'      => $expected,
                    'actual_qty'        => $actual,
                    'variance_qty'      => $expected - $actual,
                    'materials'         => $snapshot,
                    'notes'             => $notes !== '' ? $notes : null,
                    'user_id'           => auth()->id(),
                ]);

                \App\Models\FinancialAuditLog::record('production.recorded', $audit, [
                    'product'  => $product->product_name,
                    'expected' => $expected,
                    'actual'   => $actual,
                    'variance' => $expected - $actual,
                ]);

                return $audit;
            });
        } catch (\RuntimeException $e) {
            return back()->withInput()->with('error', $e->getMessage());
        } catch (\Throwable $e) {
            report($e);
            return back()->withInput()->with('error', 'Could not record production. Nothing was changed.');
        }

        $name = $audit->product_name;
        $exp  = $audit->expected_qty;
        $act  = $audit->actual_qty;

        if ($audit->variance_qty > 0) {
            return redirect()->route('production.create')->with('error',
                "VARIANCE: inventory should have produced {$exp} {$name}, but only {$act} were made ({$audit->variance_qty} short).");
        }
        if ($audit->variance_qty < 0) {
            return redirect()->route('production.create')->with('error',
                "VARIANCE: {$act} {$name} produced, " . abs($audit->variance_qty) . " MORE than the {$exp} expected.");
        }

        return redirect()->route('production.create')->with('success',
            "Recorded: {$act} {$name} produced, matches the expected {$exp}.");
    }
    /** Old "Production Audit" page now lives inside the Inventory > Production tab. */
    public function audit()
    {
        return redirect(route('production.create') . '#audit');
    }
}


===== app/Models/BranchOrder.php =====
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\DB;

class BranchOrder extends Model
{
    protected $fillable = ['destination_site_id', 'needed_by', 'status', 'notes', 'short_reason', 'created_by'];
    protected $casts = ['needed_by' => 'date'];

    public function items()
    {
        return $this->hasMany(BranchOrderItem::class);
    }

    public function destinationSite()
    {
        return $this->belongsTo(Site::class, 'destination_site_id');
    }

    /**
     * Attach delivery numbers (display only, never saved):
     * item->delivered, item->balance; order->ordered_qty, sent_qty, balance_qty.
     * Delivered = qty_sent of NON-voided batches linked to the order.
     * Balance   = ordered - delivered - closed short (never below 0).
     */
    public static function decorate($orders)
    {
        $orders = collect($orders);
        $ids = $orders->pluck('id')->all();
        $map = [];

        if (!empty($ids)) {
            $rows = DB::table('batch_items')
                ->join('batches', 'batches.id', '=', 'batch_items.batch_id')
                ->whereIn('batches.branch_order_id', $ids)
                ->whereNull('batches.voided_at')
                ->groupBy('batches.branch_order_id', 'batch_items.product_id')
                ->selectRaw('batches.branch_order_id as oid, batch_items.product_id as pid, SUM(batch_items.qty_sent) as q')
                ->get();
            foreach ($rows as $r) {
                $map[(int) $r->oid][(int) $r->pid] = (int) $r->q;
            }
        }

        foreach ($orders as $o) {
            $ord = $sent = $bal = 0;
            foreach ($o->items as $it) {
                $d = $map[(int) $o->id][(int) $it->product_id] ?? 0;
                $it->delivered = $d;
                $it->balance = max(0, (int) $it->qty - $d - (int) $it->qty_closed);
                $ord += (int) $it->qty;
                $sent += $d;
                $bal += $it->balance;
            }
            $o->ordered_qty = $ord;
            $o->sent_qty = $sent;
            $o->balance_qty = $bal;
        }

        return $orders;
    }

    /** Recompute requested/produced/partial/complete after a send or a void. */
    public function refreshStatus(): void
    {
        $fresh = static::with('items')->find($this->id);
        if (!$fresh || $fresh->status === 'cancelled') {
            return;
        }
        static::decorate(collect([$fresh]));

        if ($fresh->ordered_qty > 0 && $fresh->balance_qty <= 0) {
            $s = 'complete';
        } elseif ($fresh->sent_qty > 0) {
            $s = 'partial';
        } else {
            $s = in_array($fresh->status, ['requested', 'produced'], true) ? $fresh->status : 'requested';
        }

        if ($s !== $fresh->status) {
            static::whereKey($fresh->id)->update(['status' => $s]);
        }
    }
}

===== app/Services/FinancialStatementService.php =====
<?php

namespace App\Services;

use App\Models\BatchItem;
use App\Models\Expense;
use App\Models\Order;
use Illuminate\Support\Carbon;
use Illuminate\Support\Facades\DB;

/**
 * Single source of truth for the Financial Statement (Income Statement).
 *
 * REVENUE
 *   1. Custom orders & commissions: total_amount of orders marked Completed, dated by completed_at.
 *   2. Branch sales, by batch_date:
 *        - BEFORE the cutoff: legacy rule, (qty_sent - qty_returned) x unit_price of RECONCILED batches.
 *        - ON/AFTER the cutoff: FINAL SALE, qty_sent x unit_price of every non-voided batch.
 * EXPENSES
 *   Active (non-voided) expenses dated by expense_date.
 *   COGS = ingredients + packaging; everything else is Operating Expense.
 * GROSS PROFIT = Revenue - COGS;  NET INCOME = Gross Profit - Operating Expenses.
 */
class FinancialStatementService
{
    /** Batches dated on/after this are final sales (no returns, no reconciliation). */
    public const BRANCH_FINAL_SALE_FROM = '2026-10-07';

    public function build(Carbon $from, Carbon $to): array
    {
        $from = $from->copy()->startOfDay();
        $to   = $to->copy()->endOfDay();

        $orderRevenue = (float) Order::where('status', 'Completed')
            ->whereNotNull('completed_at')
            ->whereBetween('completed_at', [$from, $to])
            ->sum('total_amount');

        $legacyBranch = (float) BatchItem::query()
            ->join('batches', 'batches.id', '=', 'batch_items.batch_id')
            ->where('batches.status', 'reconciled')
            ->where('batches.batch_date', '<', self::BRANCH_FINAL_SALE_FROM)
            ->whereBetween('batches.batch_date', [$from->toDateString(), $to->toDateString()])
            ->sum(DB::raw('(batch_items.qty_sent - batch_items.qty_returned) * batch_items.unit_price'));

        $finalBranch = (float) BatchItem::query()
            ->join('batches', 'batches.id', '=', 'batch_items.batch_id')
            ->whereNull('batches.voided_at')
            ->where('batches.batch_date', '>=', self::BRANCH_FINAL_SALE_FROM)
            ->whereBetween('batches.batch_date', [$from->toDateString(), $to->toDateString()])
            ->sum(DB::raw('(batch_items.qty_sent - batch_items.qty_returned) * batch_items.unit_price'));

        $branchRevenue = $legacyBranch + $finalBranch;

        $byCategory = Expense::active()
            ->whereBetween('expense_date', [$from->toDateString(), $to->toDateString()])
            ->select('category', DB::raw('SUM(amount) as total'))
            ->groupBy('category')
            ->pluck('total', 'category')
            ->map(fn ($v) => (float) $v)
            ->toArray();

        $cogsLines = $opexLines = [];
        foreach ($byCategory as $cat => $amt) {
            if (in_array(strtolower($cat), Expense::COGS_CATEGORIES, true)) {
                $cogsLines[$cat] = $amt;
            } else {
                $opexLines[$cat] = $amt;
            }
        }
        ksort($cogsLines);
        ksort($opexLines);

        $revenue = round($orderRevenue + $branchRevenue, 2);
        $cogs    = round(array_sum($cogsLines), 2);
        $opex    = round(array_sum($opexLines), 2);
        $gross   = round($revenue - $cogs, 2);
        $net     = round($gross - $opex, 2);

        $voided = Expense::whereNotNull('voided_at')
            ->whereBetween('expense_date', [$from->toDateString(), $to->toDateString()]);

        return [
            'from' => $from, 'to' => $to,
            'orderRevenue' => round($orderRevenue, 2), 'branchRevenue' => round($branchRevenue, 2),
            'revenue' => $revenue,
            'cogsLines' => $cogsLines, 'cogs' => $cogs,
            'opexLines' => $opexLines, 'opex' => $opex,
            'grossProfit' => $gross, 'netIncome' => $net,
            'grossMargin' => $revenue > 0 ? round($gross / $revenue * 100, 1) : null,
            'netMargin'   => $revenue > 0 ? round($net / $revenue * 100, 1) : null,
            'voidedCount' => (clone $voided)->count(),
            'voidedTotal' => round((float) (clone $voided)->sum('amount'), 2),
            'entryCount'  => Expense::active()->whereBetween('expense_date', [$from->toDateString(), $to->toDateString()])->count(),
        ];
    }
}

        sel.value = String(pid);

function pick(sel,n){
    const o = sel.options[sel.selectedIndex];
    const price = o.dataset.price;
    const stock = parseInt(o.dataset.stock || '0', 10);
    const bal = o.dataset.bal;
    if(price) document.getElementById('price'+n).value = price;
    const qty = document.getElementById('qty'+n);
    const av = document.getElementById('avail'+n);
    if(sel.value){
        qty.max = stock;
        if(stock <= 0){ av.textContent = 'Out of stock - cannot send'; av.className='avail low'; qty.value = 0; return; }
        let q = stock;
        if(bal !== '' && bal !== undefined){ q = Math.min(parseInt(bal,10), stock); }
        qty.value = q;
        chk(n);
    } else { av.textContent=''; qty.max=''; }
}
function chk(n){
    const sel = document.getElementById('sel'+n);
    const o = sel.options[sel.selectedIndex];
    const av = document.getElementById('avail'+n);
    if(!sel.value) return;
    const stock = parseInt(o.dataset.stock || '0', 10);
    const bal = o.dataset.bal;
    const q = parseInt(document.getElementById('qty'+n).value || '0', 10);
    let msg = 'Available to send: ' + stock + ' pcs';
    let cls = 'avail ok';
    if(bal !== '' && bal !== undefined){
        const b = parseInt(bal,10);
        if(q > b){ msg += ' | Over the order by ' + (q-b); cls = 'avail low'; }
        else if(q < b){ msg += ' | Short by ' + (b-q) + ' (to be delivered in the next batch)'; }
        else { msg += ' | Matches the order'; }
    }
    if(q > stock){ msg = 'Only ' + stock + ' on hand'; cls = 'avail low'; }
    av.textContent = msg; av.className = cls;
}
function confirmSend(){
    const shorts = [];
    document.querySelectorAll('#rows tr').forEach(function(tr){
        const sel = tr.querySelector('select');
        const qty = tr.querySelector('input[type=number]');
        if(!sel || !sel.value) return;
        const o = sel.options[sel.selectedIndex];
        const bal = o.dataset.bal;
        if(bal === '' || bal === undefined) return;
        const b = parseInt(bal, 10);
        const q = parseInt(qty.value || '0', 10);
        if(q < b){ shorts.push(o.text.split(' \u2014')[0] + ' short ' + (b - q)); }
    });
    let m = 'Send this batch? It is a final sale.';
    if(shorts.length){ m += '\n\nShort: ' + shorts.join(', ') + '.\nThe shortage will be noted and delivered in the next batch.'; }
    return confirm(m);
}

    const id = parseInt(document.getElementById('orderSel').value || '0', 10);
    document.getElementById('rows').innerHTML = '';
    cur = id ? (orders.find(function(x){ return x.id === id; }) || null) : null;
    if(!cur){ addRow(); return; }
    cur.items.forEach(function(it){ if(it.balance > 0) addRow(it.product_id); });
}
loadOrder();

function opts(sel){
    const l = S.o ? avail().map(x => [x.pid, pb(x.pid).name + ' \u2014 ordered ' + x.q + ', balance ' + x.bl + ' (' + pb(x.pid).stock + ' on hand)']) : PRODUCTS.map(p => [p.id, p.name + ' (' + p.stock + ' on hand)']);
    return '<option value="">Select a product</option>' + l.map(a => '<option value="' + a[0] + '"' + (sel == a[0] ? ' selected' : '') + '>' + esc(a[1]) + '</option>').join('');
}
function drawRows(){
    $('#rws').innerHTML = S.rows.map((r, i) =>
        '<div class="brw"><div style="flex:1;min-width:0"><select class="in" name="items[' + i + '][product_id]" onchange="pk(' + i + ',this.value)" aria-label="Product">' + opts(r.pid) + '</select><div class="bh" id="h' + i + '"></div></div>'
        + '<input class="in q" type="number" min="1" name="items[' + i + '][qty_sent]" value="' + esc(r.qty) + '" oninput="qy(' + i + ',this.value)" aria-label="Quantity">'
        + '<input class="in p" type="number" step="0.01" min="0" name="items[' + i + '][unit_price]" value="' + esc(r.pr) + '" oninput="S.rows[' + i + '].pr=this.value" aria-label="Unit price">'
        + '<button type="button" class="bx" onclick="rmR(' + i + ')" aria-label="Remove row">&times;</button></div>').join('');
    S.rows.forEach(function(_, i){ hint(i); });
}
function hint(i){
    const r = S.rows[i], el = $('#h' + i); if(!el) return;
    if(!r.pid){ el.textContent = ''; return; }
    const p = pb(r.pid); if(!p){ el.textContent = ''; return; }
    const q = +r.qty || 0; let m = 'On hand: ' + p.stock, c = 'bh';
    if(q > p.stock){ m = 'Only ' + p.stock + ' on hand'; c = 'bh lo'; }
    else if(S.o){
        const it = so().items.find(x => x.pid == r.pid);
        if(it){
            if(q > it.bl){ m += ' \u00b7 Over the order by ' + (q - it.bl); c = 'bh lo'; }
            else if(q < it.bl){ m += ' \u00b7 Short by ' + (it.bl - q) + ', to be delivered in the next batch'; }
            else { m += ' \u00b7 Matches the order'; }
        }
    }
    el.textContent = m; el.className = c;
}
function resetWarn(){ S.warn = false; $('#wn').style.display = 'none'; $('#err').style.display = 'none'; $('#sb').textContent = 'Send batch'; }
function oc(v){ S.o = v || ''; if(S.o && !so()) S.o = ''; fill(); drawRows(); resetWarn(); }
function pk(i, v){
    const r = S.rows[i]; r.pid = v;
    if(v){ const p = pb(v); r.pr = p.price; const it = S.o ? so().items.find(x => x.pid == v) : null; r.qty = Math.max(1, Math.min(it ? it.bl : 1, p.stock)); }
    drawRows(); resetWarn();
}
function qy(i, v){ S.rows[i].qty = v; hint(i); resetWarn(); }
function addR(){ S.rows.push({pid:'', qty:1, pr:''}); drawRows(); }
function rmR(i){ S.rows.splice(i, 1); drawRows(); resetWarn(); }
function goSend(){
    $('#err').style.display = 'none';
    const rs = S.rows.filter(r => r.pid);
    if(!rs.length) return er('Add at least one product.');
    for(const r of rs){
        const p = pb(r.pid), q = +r.qty;
        if(!(q >= 1)) return er('Quantity must be at least 1.');
        if(q > p.stock) return er('Only ' + p.stock + ' ' + p.name + ' on hand. Lower the quantity.');
    }
    const sh = [];
    if(S.o) avail().forEach(function(x){
        const s = rs.filter(r => r.pid == x.pid).reduce((a, r) => a + (+r.qty), 0);
        if(s < x.bl) sh.push(pb(x.pid).name + ' short ' + (x.bl - s));
    });
    if(sh.length && !S.warn){
        S.warn = true; const w = $('#wn');
        w.innerHTML = '<b>Short: ' + esc(sh.join(', ')) + '.</b> You can still send. The shortage is noted and delivered in the next batch.';
        w.style.display = 'block'; $('#sb').textContent = 'Send anyway'; return;
    }
    S.rows = rs; drawRows(); $('#bf').submit();
}

(function(){
  var PROD = @json($yield);
  function el(id){ return document.getElementById(id); }
  function esc(s){ return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;'); }
  function f(n){ return (Math.round(Number(n)*100)/100).toLocaleString(undefined,{maximumFractionDigits:2}); }

  window.openProd = function(id){
    el('prodBg').classList.add('show');
    if(id){ el('pProd').value = String(id); }

    el('prodInfo').style.display = p ? 'block' : 'none';
    if(p){ el('pMax').textContent = p.max; el('pExp').max = p.max; }
    window.recalc();
  };
  window.useMax = function(){
    var p = PROD[el('pProd').value];
    if(p){ el('pExp').value = p.max; window.recalc(); }
  };
  window.recalc = function(){
    var p = PROD[el('pProd').value];
    var btn = el('pSubmit'), box = el('pVar');
    box.className = 'pm-var'; box.textContent = '';
    if(!p){ btn.disabled = true; return; }

    var e = parseInt(el('pExp').value, 10), a = parseInt(el('pAct').value, 10);
    var eN = isNaN(e) ? 0 : e;
    var rows = '';
    p.items.forEach(function(it){
      var need = it.per * eN, after = it.on - need;
      rows += '<tr><td>'+esc(it.name)+'</td><td>'+f(it.on)+' '+esc(it.unit)+'</td><td>'+f(it.per)+'</td><td>'+(eN ? f(need) : '-')+'</td><td class="'+(eN && after < 0 ? 'neg' : '')+'">'+(eN ? f(after) : '-')+'</td></tr>';
    });
    el('pMatBody').innerHTML = rows;

    var valid = !isNaN(e) && e >= 1 && e <= p.max && !isNaN(a) && a >= 0;
    if(!isNaN(e) && e > p.max){
      box.className = 'pm-var bad';
      box.textContent = 'Not enough materials: stock only covers up to ' + p.max + '.';
    } else if(valid){
      var d = e - a;
      if(d > 0){
        box.className = 'pm-var bad';
        box.textContent = 'Materials were issued for ' + e + ', but only ' + a + ' produced: ' + d + ' SHORT. Where did they go?';
      } else if(d < 0){
        box.className = 'pm-var warn';
        box.textContent = 'Produced ' + a + ' which is ' + Math.abs(d) + ' MORE than the ' + e + ' the materials should give. Please re-check the count.';
      } else {
        box.className = 'pm-var ok';
        box.textContent = 'Match: ' + a + ' produced, as expected.';
      }
      if(d !== 0 && el('pNotes').value.trim() === ''){ valid = false; }
    }
    btn.disabled = !valid;
  };

  document.getElementById('prodBg').addEventListener('click', function(ev){ if(ev.target === this){ window.closeProd(); } });

  @if(old('product_id'))

<script>
const cart = new Map(); // id -> {id,name,price,stock,qty}
var posCurCat = "";

function peso(n){ return "\u20B1" + Number(n).toLocaleString("en-PH",{minimumFractionDigits:2,maximumFractionDigits:2}); }
function cartTotal(){ let t=0; cart.forEach(i=> t += i.price*i.qty); return Math.round(t*100)/100; }

function posAdd(btn){
  const id = btn.dataset.id;
  const stock = parseInt(btn.dataset.stock,10);
  const cur = cart.get(id);
  if (cur){ if (cur.qty < stock) cur.qty++; }
  else { cart.set(id, {id, name:btn.dataset.name, price:parseFloat(btn.dataset.price), stock, qty:1}); }
  renderCart();
}
function posSet(id, d){
  const i = cart.get(id); if(!i) return;
  i.qty += d;
  if (i.qty <= 0) cart.delete(id);
  else if (i.qty > i.stock) i.qty = i.stock;
  renderCart();
}
function posRemove(id){ cart.delete(id); renderCart(); }

function renderCart(){
  const body = document.getElementById("cartBody");
  const empty = document.getElementById("cartEmpty");
  body.querySelectorAll(".cline").forEach(e=>e.remove());
  if (cart.size === 0){ empty.style.display="block"; }
  else {
    empty.style.display="none";
    cart.forEach(i=>{
      const row = document.createElement("div");
      row.className="cline";
      row.innerHTML =
        '<div class="ci"><div class="cn">'+i.name+'</div><div class="cu">'+peso(i.price)+' each</div></div>'+
        '<button type="button" class="qbtn" onclick="posSet(\''+i.id+'\',-1)">&minus;</button>'+
        '<span class="qn">'+i.qty+'</span>'+
        '<button type="button" class="qbtn" onclick="posSet(\''+i.id+'\',1)">+</button>'+
        '<span class="clt">'+peso(i.price*i.qty)+'</span>'+
        '<button type="button" class="crm" title="Remove" onclick="posRemove(\''+i.id+'\')">&times;</button>';
      body.appendChild(row);
    });
  }
  document.getElementById("cartTotal").textContent = peso(cartTotal());
  document.getElementById("chargeBtn").disabled = cart.size === 0;
}

function posSetCat(btn){
  document.querySelectorAll("#posChips .pchip").forEach(c=>c.classList.remove("active"));
  btn.classList.add("active");
  posCurCat = btn.dataset.cat || "";
  posFilter();
}
function posFilter(){
  const q = document.getElementById("posSearch").value.toLowerCase().trim();
  let shown = 0;
  document.querySelectorAll("#posSections .cat-sec").forEach(sec=>{
    const catOk = !posCurCat || sec.dataset.cat === posCurCat;
    let n = 0;
    sec.querySelectorAll(".pcard").forEach(c=>{
      const show = catOk && c.dataset.name.toLowerCase().includes(q);
      c.style.display = show ? "" : "none";
      if (show) { n++; shown++; }
    });
    sec.style.display = n ? "" : "none";
  });
  document.getElementById("posNone").style.display = shown ? "none" : "block";
}

/* ----- payment modal ----- */
function posOpenPay(){
  if (cart.size === 0) return;
  document.getElementById("payDue").textContent = peso(cartTotal());
  document.getElementById("cashAmt").value = "";
  document.getElementById("onlineMethod").value = "";
  document.getElementById("onlineAmt").value = "";
  document.getElementById("onlineRef").value = "";
  posRecalc();
  document.getElementById("payBg").classList.add("show");
}
function posClosePay(){ document.getElementById("payBg").classList.remove("show"); }

function posExact(){
  const total  = cartTotal();
  const method = document.getElementById("onlineMethod").value;
  const oAmt   = method ? (parseFloat(document.getElementById("onlineAmt").value) || 0) : 0;
  const due    = Math.max(0, Math.round((total - oAmt) * 100) / 100);
  document.getElementById("cashAmt").value = due.toFixed(2);
  posRecalc();
}

function posRecalc(){
  const total  = cartTotal();
  const cash   = parseFloat(document.getElementById("cashAmt").value) || 0;
  const method = document.getElementById("onlineMethod").value;
  const oAmt   = parseFloat(document.getElementById("onlineAmt").value) || 0;
  const oRef   = document.getElementById("onlineRef").value.trim();

  const box = document.getElementById("onlineBox");
  if (method) box.classList.add("show"); else box.classList.remove("show");

  const paid = Math.round((cash + (method ? oAmt : 0)) * 100) / 100;
  const rem  = Math.round((total - paid) * 100) / 100;

  document.getElementById("paySum").textContent = peso(paid);
  const remRow = document.getElementById("payRemRow");
  if (rem > 0.009){
    document.getElementById("payRemLbl").textContent = "Remaining";
    document.getElementById("payRem").textContent = peso(rem);
    remRow.classList.add("short"); remRow.classList.remove("ok");
  } else {
    document.getElementById("payRemLbl").textContent = rem < -0.009 ? "Change (give manually)" : "Fully paid";
    document.getElementById("payRem").textContent = peso(Math.abs(rem));
    remRow.classList.add("ok"); remRow.classList.remove("short");
  }

  const refMissing = method && oAmt > 0 && !oRef;
  document.getElementById("payConfirm").disabled = (paid + 0.009 < total) || refMissing || cart.size === 0;
}

document.getElementById("posForm").addEventListener("submit", function(e){
  const box = document.getElementById("posHidden"); box.innerHTML = "";

    n++;
  });
  let m = 0;
  const cash = parseFloat(document.getElementById("cashAmt").value) || 0;
  if (cash > 0){
    box.insertAdjacentHTML("beforeend",
      '<input type="hidden" name="payments['+m+'][method]" value="cash">'+
      '<input type="hidden" name="payments['+m+'][amount]" value="'+cash.toFixed(2)+'">'+
      '<input type="hidden" name="payments['+m+'][reference]" value="">');
    m++;
  }
  const method = document.getElementById("onlineMethod").value;
  const oAmt   = parseFloat(document.getElementById("onlineAmt").value) || 0;
  const oRef   = document.getElementById("onlineRef").value.trim();
  if (method && oAmt > 0){
    box.insertAdjacentHTML("beforeend",
      '<input type="hidden" name="payments['+m+'][method]" value="'+method+'">'+
      '<input type="hidden" name="payments['+m+'][amount]" value="'+oAmt.toFixed(2)+'">'+
      '<input type="hidden" name="payments['+m+'][reference]" value="'+oRef+'">');
    m++;
  }
});

    <script>
        const RECIPES = @json($recipes);
        function fmt(n){ n = Number(n) || 0; return (Math.round(n*100)/100).toString(); }
        function renderRecipe(){
            const pid = document.getElementById('prodSelect').value;
            const qty = parseInt(document.getElementById('prodQty').value, 10) || 0;
            const wrap = document.getElementById('ingWrap');
            const sub = document.getElementById('prodSubmit');
            const warn = document.getElementById('prodWarn');
            const r = RECIPES[pid];
            if (!r){ wrap.innerHTML = ''; sub.disabled = true; warn.style.display='none'; return; }

            let maxMake = null;
            r.items.forEach(function(it){ if (it.qty_used > 0){ const c = Math.floor(it.stock / it.qty_used); maxMake = (maxMake===null) ? c : Math.min(maxMake, c); } });

            let html = '<div class="pr-max">Max producible with current stock: <strong>' + (maxMake===null ? '-' : maxMake) + '</strong></div>';
            html += '<table class="pr-table"><thead><tr><th>Ingredient</th><th>Stock before</th><th>Per unit</th><th>Expected use</th><th>Expected after</th><th>Actual remaining</th></tr></thead><tbody>';
            r.items.forEach(function(it){
                const expected = Math.round(it.qty_used * qty * 100) / 100;
                const expAfter = Math.round((it.stock - expected) * 100) / 100;
                const low = expAfter < -0.0001;
                html += '<tr' + (low ? ' class="pr-bad"' : '') + '>' +
                    '<td>' + it.name + '</td>' +
                    '<td>' + fmt(it.stock) + ' ' + it.unit + '</td>' +
                    '<td>' + fmt(it.qty_used) + '</td>' +
                    '<td>' + fmt(expected) + ' ' + it.unit + '</td>' +
                    '<td>' + fmt(expAfter) + ' ' + it.unit + '</td>' +
                    '<td><input type="number" step="0.01" min="0" name="actual[' + it.inventory_id + ']" placeholder="' + fmt(expAfter) + '" class="pr-actual"></td>' +
                    '</tr>';
            });
            html += '</tbody></table>';
            html += '<p class="pr-hint">Leave <strong>Actual remaining</strong> blank to trust the recipe. Enter the real counted amount to audit for variance (theft / waste).</p>';
            wrap.innerHTML = html;

            const over = (maxMake !== null && qty > maxMake);
            warn.style.display = over ? 'block' : 'none';
            sub.disabled = !(pid && qty > 0 && !over);
        }
        document.addEventListener('DOMContentLoaded', renderRecipe);
    </script>
</x-app-layout>

function previewImg(input){
    var img = document.getElementById('preview');
    var col = document.getElementById('newCol');
    var arrow = document.getElementById('photoArrow');
    var err = document.getElementById('imgErr');
    err.style.display = 'none';
    col.style.display = 'none'; arrow.style.display = 'none';

    if(!(input.files && input.files[0])){ return; }
    var f = input.files[0];
    if(f.size > 2 * 1024 * 1024){
        err.textContent = 'This file is ' + (f.size / 1048576).toFixed(1) + 'MB. Maximum is 2MB. Please choose a smaller photo.';
        err.style.display = 'block';
        input.value = '';
        return;
    }
    img.src = URL.createObjectURL(f);
    col.style.display = 'block'; arrow.style.display = 'block';
}

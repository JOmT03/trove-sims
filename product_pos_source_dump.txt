===== app/Http/Controllers/ProductController.php =====
<?php

namespace App\Http\Controllers;

use App\Models\Product;
use App\Models\Inventory;
use App\Models\InventoryLog;
use App\Models\Site;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;

class ProductController extends Controller
{
    public function index()
    {
        $products = Product::with('materials')->orderBy('created_at', 'desc')->get();
        return view('products.index', compact('products'));
    }

    public function create()
    {
        $inventoryItems = Inventory::orderBy('item_name')->get();
        $sites = Site::all();
        return view('products.create', compact('inventoryItems', 'sites'));
    }

    public function store(Request $request)
    {
        try {
            $validated = $request->validate([
                'product_name'  => 'required|string|max:255|unique:products',
                'description'   => 'nullable|string|max:1000',
                'category'      => 'nullable|string|max:255',
                'price'         => 'required|numeric|min:0',
                'stock_quantity' => 'nullable|numeric|min:0',
                'site_id'       => 'nullable|exists:sites,id',
                'image'         => 'nullable|image|max:2048',
                'recipe'        => 'nullable|array',
                'recipe.*.inventory_id' => 'numeric|exists:inventory,id',
                'recipe.*.quantity_needed' => 'numeric|min:0.01',
            ]);

            DB::beginTransaction();

            $imagePath = null;
            if ($request->hasFile('image')) {
                $imagePath = $request->file('image')->store('products', 'public');
            }

            $qtyToMake = (float) ($validated['stock_quantity'] ?? 0);

            $product = Product::create([
                'product_name'   => $validated['product_name'],
                'description'    => $validated['description'] ?? null,
                'category'       => $validated['category'] ?? null,
                'price'          => $validated['price'],
                'stock_quantity' => $qtyToMake,
                'site_id'        => $validated['site_id'] ?? null,
                'status'         => 'active',
                'image_path'     => $imagePath,
            ]);

            if (! empty($validated['recipe'])) {
                $materials = [];
                foreach ($validated['recipe'] as $recipe_item) {
                    if (empty($recipe_item['inventory_id'])) continue;

                    $inventory_id = $recipe_item['inventory_id'];
                    $perUnit      = (float) $recipe_item['quantity_needed'];
                    $inventory    = Inventory::findOrFail($inventory_id);

                    // Deduct for the whole starting batch (recipe x quantity produced now)
                    $totalNeeded = $perUnit * $qtyToMake;

                    if ($totalNeeded > 0) {
                        if ($inventory->quantity_on_hand < $totalNeeded) {
                            throw new \Exception(
                                "Insufficient {$inventory->item_name}. Available: {$inventory->quantity_on_hand}, Needed: {$totalNeeded}"
                            );
                        }

                        $inventory->update([
                            'quantity_on_hand' => $inventory->quantity_on_hand - $totalNeeded
                        ]);

                        InventoryLog::create([
                            'inventory_id' => $inventory_id,
                            'type'         => 'used',
                            'quantity'     => $totalNeeded,
                            'reference'    => 'PRODUCT-CREATE',
                            'notes'        => "Produced {$qtyToMake} x {$product->product_name} on create",
                            'user_id'      => auth()->id(),
                        ]);
                    }

                    // Store the per-unit recipe (mirror for the active recipe / Recipes page)
                    $materials[$inventory_id] = ['quantity_used' => $perUnit];
                }

                if (! empty($materials)) {
                    $product->materials()->attach($materials);
                }
            }

            DB::commit();

            $msg = $qtyToMake > 0
                ? "Product '{$product->product_name}' created and {$qtyToMake} pcs produced. Ingredients deducted from inventory."
                : "Product '{$product->product_name}' created. You can produce a batch anytime from its Recipes page.";

            return redirect()->route('products.index')->with('success', $msg);
        } catch (\Exception $e) {
            DB::rollBack();
            \Log::error('Product creation failed: ' . $e->getMessage());
            return back()->withInput()
                ->with('error', 'Failed to create product: ' . $e->getMessage());
        }
    }

    public function show(Product $product)
    {
        $product->load('materials');
        return view('products.show', compact('product'));
    }

    public function edit(Product $product)
    {
        $inventoryItems = Inventory::orderBy('item_name')->get();
        $sites = Site::all();
        $product->load('materials');
        return view('products.edit', compact('product', 'inventoryItems', 'sites'));
    }

    public function update(Request $request, Product $product)
    {
        try {
            $validated = $request->validate([
                'product_name'  => 'required|string|max:255|unique:products,product_name,' . $product->id,
                'description'   => 'nullable|string|max:1000',
                'category'      => 'nullable|string|max:255',
                'price'         => 'required|numeric|min:0',
                'site_id'       => 'nullable|exists:sites,id',
                'status'        => 'nullable|in:active,inactive',
                'image'         => 'nullable|image|max:2048',
            ]);

            if ($request->hasFile('image')) {
                if ($product->image_path && Storage::disk('public')->exists($product->image_path)) {
                    Storage::disk('public')->delete($product->image_path);
                }
                $validated['image_path'] = $request->file('image')->store('products', 'public');
            }

            unset($validated['image']);
            $product->update($validated);

            return redirect()->route('products.index')
                ->with('success', 'Product updated successfully.');
        } catch (\Exception $e) {
            \Log::error('Product update failed: ' . $e->getMessage());
            return back()->withInput()
                ->with('error', 'Failed to update product. Please try again.');
        }
    }

    public function destroy(Product $product)
    {
        try {
            if ($product->image_path && Storage::disk('public')->exists($product->image_path)) {
                Storage::disk('public')->delete($product->image_path);
            }
            $product->delete();
            return redirect()->route('products.index')
                ->with('success', 'Product deleted successfully.');
        } catch (\Exception $e) {
            \Log::error('Product deletion failed: ' . $e->getMessage());
            return back()->with('error', 'Failed to delete product. Please try again.');
        }
    }
}

<script>
const inventoryItems = {!! $invData->toJson() !!};
let rowIndex = 0;

function previewImg(input){
    const img = document.getElementById('preview');
    if(input.files && input.files[0]){ img.src = URL.createObjectURL(input.files[0]); img.style.display='block'; }
    else { img.style.display='none'; }
}
function itemOptions(){
    return inventoryItems.map(function(i){ return '<option value="'+i.id+'">'+i.name+' ('+i.unit+')</option>'; }).join('');
}
function invById(id){ return inventoryItems.find(function(i){ return String(i.id)===String(id); }); }

function addRow(){
    const i = rowIndex++;
    const tr = document.createElement('tr');
    tr.id = 'recipe_row_'+i;
    tr.innerHTML =
        '<td><select name="recipe['+i+'][inventory_id]" onchange="recalc()" required><option value="">- Select item -</option>'+itemOptions()+'</select></td>' +
        '<td><input type="number" name="recipe['+i+'][quantity_needed]" step="0.01" min="0.01" oninput="recalc()" required></td>' +
        '<td><button type="button" onclick="document.getElementById(\'recipe_row_'+i+'\').remove();recalc();" style="background:none;border:none;color:#ef4444;cursor:pointer;font-size:16px;">&times;</button></td>';
    document.getElementById('recipeBody').appendChild(tr);
    recalc();
}

function recalc(){
    const rows = document.querySelectorAll('#recipeBody tr');
    const box = document.getElementById('canmake');
    const warn = document.getElementById('warn');
    let valid = 0, maxMake = Infinity;
    rows.forEach(function(r){
        const sel = r.querySelector('select');
        const inp = r.querySelector('input');
        const id = sel ? sel.value : '';
        const qty = inp ? parseFloat(inp.value) : NaN;
        if(id && qty > 0){ const it = invById(id); if(it){ valid++; maxMake = Math.min(maxMake, Math.floor(it.on / qty)); } }
    });
    warn.style.display = 'none';
    if(valid === 0){ box.className='canmake none'; box.innerHTML='Add at least one ingredient to see how many you can make.'; return; }
    if(maxMake <= 0){ box.className='canmake zero'; box.innerHTML='&#9888; Not enough stock to make even 1 unit with this recipe. Restock first.'; }
    else { box.className='canmake ok'; box.innerHTML='With current stock, you can make up to <span class="big">'+maxMake.toLocaleString('en-PH')+'</span> pcs of this product.'; }
    const sq = document.getElementById('stockq');
    const start = sq ? (parseFloat(sq.value) || 0) : 0;
    if(maxMake !== Infinity && start > maxMake){
        warn.style.display='block';
        warn.innerHTML='&#9888; You only have enough stock for '+maxMake.toLocaleString('en-PH')+' pcs. Lower the amount or restock first.';
    }
}
function toggleInv(){
    document.getElementById('invlist').classList.toggle('hide');
    document.getElementById('invchev').classList.toggle('closed');
}
addRow();
</script>

    document.querySelectorAll('#filterMenu button').forEach(function(b){ b.classList.remove('active'); });

    document.getElementById('filterMenu').hidden = true; applyFilters();
}
function applyFilters(){
    var term = (document.getElementById('search').value || '').toLowerCase().trim();
    var total = 0;
    document.querySelectorAll('.cat-sec').forEach(function(sec){
        var catOk = curCat === 'all' || sec.getAttribute('data-cat') === curCat;
        var shown = 0;
        sec.querySelectorAll('.p-card').forEach(function(c){
            var okName = c.getAttribute('data-name').indexOf(term) !== -1;
            var state  = c.getAttribute('data-state');
            var okStat = curStatus === 'all' || (curStatus === 'in' && state === 'in') || (curStatus === 'low' && (state === 'low' || state === 'out'));
            var show = catOk && okName && okStat;
            c.hidden = !show; if(show){ shown++; total++; }
        });
        sec.hidden = shown === 0;
    });
    var nr = document.getElementById('noresult'); if(nr) nr.style.display = total === 0 ? 'block' : 'none';
}

<?php

namespace App\Http\Controllers;

use App\Models\Product;
use App\Models\Sale;
use App\Models\SalePayment;
use App\Models\DayClosure;
use App\Models\Site;
use App\Models\FinancialAuditLog;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

/**
 * POS â€” counter sale of FINISHED products (walk-in).
 * Deducts finished stock (Product.stock_quantity) only. Raw materials are NOT
 * touched here: they were already consumed when the product was produced.
 * A sale is a ledger entry â€” it is voided (with a reason), never deleted.
 */
class PosController extends Controller
{
    public function index()
    {
        $products = Product::where('status', 'active')
            ->orderBy('product_name')
            ->get(['id', 'product_name', 'category', 'price', 'stock_quantity']);

        $recent = Sale::with('cashier')
            ->orderByDesc('id')
            ->limit(8)
            ->get();

        return view('pos.index', compact('products', 'recent'));
    }

    public function store(Request $request)
    {
        $data = $request->validate([
            'items'                => 'required|array|min:1',
            'items.*.product_id'   => 'required|exists:products,id',
            'items.*.quantity'     => 'required|integer|min:1',
            'payments'             => 'required|array|min:1',
            'payments.*.method'    => 'required|in:' . implode(',', SalePayment::METHODS),
            'payments.*.amount'    => 'required|numeric|min:0.01',
            'payments.*.reference' => 'nullable|string|max:100',
        ]);

        // Digital payments must carry a reference number.
        foreach ($data['payments'] as $p) {
            if ($p['method'] !== 'cash' && blank($p['reference'] ?? null)) {
                return back()->withInput()
                    ->with('error', 'A reference number is required for ' . strtoupper($p['method']) . ' payments.');
            }
        }

        try {
            $sale = DB::transaction(function () use ($data) {
                // Lock the products we are about to sell so two cashiers cannot
                // oversell the same stock at the same time.
                $ids = collect($data['items'])->pluck('product_id')->all();
                $products = Product::whereIn('id', $ids)->lockForUpdate()->get()->keyBy('id');

                $total = 0;
                $lines = [];
                foreach ($data['items'] as $row) {
                    $product = $products[$row['product_id']];
                    $qty = (int) $row['quantity'];
                    if ((float) $product->stock_quantity < $qty) {
                        throw new \RuntimeException(
                            "Not enough stock for {$product->product_name} (have {$product->stock_quantity}, need {$qty})."
                        );
                    }
                    $lineTotal = round((float) $product->price * $qty, 2);
                    $total += $lineTotal;
                    $lines[] = [$product, $qty, $lineTotal];
                }

                $paid = round(collect($data['payments'])->sum(fn ($p) => (float) $p['amount']), 2);
                if ($paid + 0.009 < $total) {
                    throw new \RuntimeException(
                        'Payments (' . number_format($paid, 2) . ') are less than the total (' . number_format($total, 2) . ').'
                    );
                }

                $siteId = auth()->user()->site_id ?? Site::query()->value('id');

                $sale = Sale::create([
                    'sale_number'  => 'S-' . now()->format('ymd') . '-' . strtoupper(substr(uniqid(), -5)),
                    'site_id'      => $siteId,
                    'cashier_id'   => auth()->id(),
                    'total'        => $total,
                    'status'       => 'completed',
                    'completed_at' => now(),
                ]);

                foreach ($lines as [$product, $qty, $lineTotal]) {
                    $sale->items()->create([
                        'product_id'   => $product->id,
                        'product_name' => $product->product_name,
                        'quantity'     => $qty,
                        'unit_price'   => $product->price,
                        'line_total'   => $lineTotal,
                    ]);
                    $product->decrement('stock_quantity', $qty); // finished stock only
                }

                foreach ($data['payments'] as $p) {
                    if ((float) $p['amount'] <= 0) {
                        continue;
                    }
                    $sale->payments()->create([
                        'method'    => $p['method'],
                        'amount'    => round((float) $p['amount'], 2),
                        'reference' => $p['method'] === 'cash' ? null : ($p['reference'] ?? null),
                    ]);
                }

                FinancialAuditLog::record('sale.recorded', $sale, [
                    'total'    => (string) $sale->total,
                    'items'    => count($lines),
                    'payments' => collect($data['payments'])
                        ->map(fn ($p) => $p['method'] . ':' . $p['amount'])->implode(', '),
                ]);

                return $sale;
            });
        } catch (\Throwable $e) {
            return back()->withInput()->with('error', $e->getMessage());
        }

        return redirect()->route('pos.index')
            ->with('success', 'Sale ' . $sale->sale_number . ' recorded. Total: ' . number_format($sale->total, 2) . '.');
    }

    public function void(Request $request, Sale $sale)
    {
        $data = $request->validate(['reason' => 'required|string|max:255']);

        if ($sale->isVoided()) {
            return back()->with('error', 'That sale is already voided.');
        }
        if (DayClosure::isClosed($sale->completed_at)) {
            return back()->with('error', 'That sale falls on a day that is already closed. It can no longer be voided.');
        }

        DB::transaction(function () use ($sale, $data) {
            foreach ($sale->items as $item) {
                if ($item->product_id) {
                    $product = Product::where('id', $item->product_id)->lockForUpdate()->first();
                    $product?->increment('stock_quantity', $item->quantity); // put the stock back
                }
            }

            $sale->update([
                'status'      => 'voided',
                'voided_at'   => now(),
                'voided_by'   => auth()->id(),
                'void_reason' => $data['reason'],
            ]);

            FinancialAuditLog::record('sale.voided', $sale, [
                'reason' => $data['reason'],
                'total'  => (string) $sale->total,
            ]);
        });

        return back()->with('success', 'Sale ' . $sale->sale_number . ' voided and stock restored.');
    }
}

<script>
const cart = new Map(); // id -> {id,name,price,stock,qty}

function peso(n){ return "\u20B1" + Number(n).toLocaleString("en-PH",{minimumFractionDigits:2,maximumFractionDigits:2}); }
function cartTotal(){ let t=0; cart.forEach(i=> t += i.price*i.qty); return Math.round(t*100)/100; }

function posAdd(btn){
  const id = btn.dataset.id;
  const stock = parseInt(btn.dataset.stock,10);
  const cur = cart.get(id);
  if (cur){ if (cur.qty < stock) cur.qty++; }
  else { cart.set(id, {id, name:btn.dataset.name, price:parseFloat(btn.dataset.price), stock, qty:1}); }
  renderCart();
}
function posSet(id, d){
  const i = cart.get(id); if(!i) return;
  i.qty += d;
  if (i.qty <= 0) cart.delete(id);
  else if (i.qty > i.stock) i.qty = i.stock;
  renderCart();
}
function posRemove(id){ cart.delete(id); renderCart(); }

function renderCart(){
  const body = document.getElementById("cartBody");
  const empty = document.getElementById("cartEmpty");
  body.querySelectorAll(".cline").forEach(e=>e.remove());
  if (cart.size === 0){ empty.style.display="block"; }
  else {
    empty.style.display="none";
    cart.forEach(i=>{
      const row = document.createElement("div");
      row.className="cline";
      row.innerHTML =
        '<div class="ci"><div class="cn">'+i.name+'</div><div class="cu">'+peso(i.price)+' each</div></div>'+
        '<button type="button" class="qbtn" onclick="posSet(\''+i.id+'\',-1)">&minus;</button>'+
        '<span class="qn">'+i.qty+'</span>'+
        '<button type="button" class="qbtn" onclick="posSet(\''+i.id+'\',1)">+</button>'+
        '<span class="clt">'+peso(i.price*i.qty)+'</span>'+
        '<button type="button" class="crm" title="Remove" onclick="posRemove(\''+i.id+'\')">&times;</button>';
      body.appendChild(row);
    });
  }
  document.getElementById("cartTotal").textContent = peso(cartTotal());
  document.getElementById("chargeBtn").disabled = cart.size === 0;
}

function posFilter(){
  const q = document.getElementById("posSearch").value.toLowerCase();
  const cat = document.getElementById("posCat").value;
  let shown = 0;
  document.querySelectorAll("#posGrid .pcard").forEach(c=>{
    const okq = c.dataset.name.toLowerCase().includes(q);
    const okc = !cat || c.dataset.cat === cat;
    const show = okq && okc; c.style.display = show ? "" : "none"; if(show) shown++;
  });
  document.getElementById("posNone").style.display = shown ? "none" : "block";
}

/* ----- payment modal ----- */
function posOpenPay(){
  if (cart.size === 0) return;
  document.getElementById("payDue").textContent = peso(cartTotal());
  document.getElementById("cashAmt").value = "";
  document.getElementById("onlineMethod").value = "";
  document.getElementById("onlineAmt").value = "";
  document.getElementById("onlineRef").value = "";
  posRecalc();
  document.getElementById("payBg").classList.add("show");
}
function posClosePay(){ document.getElementById("payBg").classList.remove("show"); }

function posExact(){
  const total  = cartTotal();
  const method = document.getElementById("onlineMethod").value;
  const oAmt   = method ? (parseFloat(document.getElementById("onlineAmt").value) || 0) : 0;
  const due    = Math.max(0, Math.round((total - oAmt) * 100) / 100);
  document.getElementById("cashAmt").value = due.toFixed(2);
  posRecalc();
}

function posRecalc(){
  const total  = cartTotal();
  const cash   = parseFloat(document.getElementById("cashAmt").value) || 0;
  const method = document.getElementById("onlineMethod").value;
  const oAmt   = parseFloat(document.getElementById("onlineAmt").value) || 0;
  const oRef   = document.getElementById("onlineRef").value.trim();

  const box = document.getElementById("onlineBox");
  if (method) box.classList.add("show"); else box.classList.remove("show");

  const paid = Math.round((cash + (method ? oAmt : 0)) * 100) / 100;
  const rem  = Math.round((total - paid) * 100) / 100;

  document.getElementById("paySum").textContent = peso(paid);
  const remRow = document.getElementById("payRemRow");
  if (rem > 0.009){
    document.getElementById("payRemLbl").textContent = "Remaining";
    document.getElementById("payRem").textContent = peso(rem);
    remRow.classList.add("short"); remRow.classList.remove("ok");
  } else {
    document.getElementById("payRemLbl").textContent = rem < -0.009 ? "Change (give manually)" : "Fully paid";
    document.getElementById("payRem").textContent = peso(Math.abs(rem));
    remRow.classList.add("ok"); remRow.classList.remove("short");
  }

  const refMissing = method && oAmt > 0 && !oRef;
  document.getElementById("payConfirm").disabled = (paid + 0.009 < total) || refMissing || cart.size === 0;
}

document.getElementById("posForm").addEventListener("submit", function(e){
  const box = document.getElementById("posHidden"); box.innerHTML = "";
  let n = 0;
  cart.forEach(i=>{
    box.insertAdjacentHTML("beforeend",
      '<input type="hidden" name="items['+n+'][product_id]" value="'+i.id+'">'+
      '<input type="hidden" name="items['+n+'][quantity]" value="'+i.qty+'">');
    n++;
  });
  let m = 0;
  const cash = parseFloat(document.getElementById("cashAmt").value) || 0;
  if (cash > 0){
    box.insertAdjacentHTML("beforeend",
      '<input type="hidden" name="payments['+m+'][method]" value="cash">'+
      '<input type="hidden" name="payments['+m+'][amount]" value="'+cash.toFixed(2)+'">'+
      '<input type="hidden" name="payments['+m+'][reference]" value="">');
    m++;
  }
  const method = document.getElementById("onlineMethod").value;
  const oAmt   = parseFloat(document.getElementById("onlineAmt").value) || 0;
  const oRef   = document.getElementById("onlineRef").value.trim();
  if (method && oAmt > 0){
    box.insertAdjacentHTML("beforeend",
      '<input type="hidden" name="payments['+m+'][method]" value="'+method+'">'+
      '<input type="hidden" name="payments['+m+'][amount]" value="'+oAmt.toFixed(2)+'">'+
      '<input type="hidden" name="payments['+m+'][reference]" value="'+oRef+'">');
    m++;
  }
});
